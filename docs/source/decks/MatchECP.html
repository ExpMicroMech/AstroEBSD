Match the pattern

%prepare the pattern

file_list=linspace(0,360,73); %0 to 360, with 73 patterns in between as per the stack
num_files=size(file_list,2);
for n=1:num_files

    Input_Data_n=Input_Data;
    Input_Data_n.image_name=[int2str(file_list(n)) '.tif'];

% Experimental ECP pattern load
[ECP_Pat] = AstroECP_Load(Input_Data_n);

[ pattern1 ] = EBSP_BGCor( ECP_Pat.ECP_Pat_BG, Settings_CorX);
% [ pattern1 ] = EBSP_BGCor( thisProcdPattern, []);
[Pat_Ref_r,XCF_data_fill] = refine_prep(pattern1,SettingsXCF,RTM);

usePar=1; %use the parallel loop for the library testing
[G_SO3,Library_PH,~,~]=fLibraryTest(template_library,library_G,Pat_Ref_r.FFT,SettingsXCF,XCF_data_fill,usePar);
%refine based upon the SO3 found solution
[G_Refined_SO3,regout_RSO3] = refine5(Pat_Ref_r,EBSD_geom,EBSD_geom.PC,G_SO3,SettingsXCF,screen_int,RTM_info.isHex,RTM);
 toc

Show the result

RTM_out=RTM;
RTM_out.screensize=ECP_Pat.size(1);
[screen_int_out] = Cube_Generate(RTM_info.bin_file,RTM_info.isHex);

[EBSD_geom_out ] = EBSP_Gnom( RTM_out, Input_Data.PC_in); %

%generate the simulated patterns to check solutions
[ Pat_sim_SO3 ] = EBSP_gen( EBSD_geom_out,G_SO3,screen_int_out); %generate the EBSP for this iteration
[ Pat_sim_SO3_refined ] = EBSP_gen( EBSD_geom_out,G_Refined_SO3,screen_int_out); %generate the EBSP for this iteration

if isdir(resultsdir) == 0; mkdir(resultsdir); end
