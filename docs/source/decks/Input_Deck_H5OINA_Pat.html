Simulate the pattern

%simulate this pattern
Input_Data.Phase_Input{1}='Si_20kv_HR';
Input_Data.Phase_Folder = fullfile(Input_Data.astro_location,'phases'); %location of the AstroEBSD phases super-folder
Input_Data.Bin_loc = fullfile(Input_Data.Phase_Folder,'dynamic_templates'); %location of the binary files used for RTM

%build the phase
[ Crystal_UCell,Crystal_Family,Crystal_LUT,Settings_LUT,num_Phases, RTM_info ] = Phase_Builder_RTM(  {Input_Data.Phase_Input{1}},Input_Data.Phase_Folder);
[screen_int] = Cube_Generate(RTM_info.bin_file,RTM_info.isHex);

%create the rotation matricies we use
Input_Data.Rz=@(theta)[cos(theta) sin(theta) 0;-sin(theta) cos(theta) 0;0 0 1]; %z rotation
Input_Data.Rx=@(theta)[1 0 0;0 cos(theta) sin(theta);0 -sin(theta) cos(theta)]; %x rotation
Input_Data.Ry=@(theta)[cos(theta) 0 sin(theta);0 1 0; -sin(theta) 0 cos(theta)]; %y rotation

%create the detector rotations required
detector_Euler=double(dataset_header.Detector_Orientation_Euler);
g_camera=Input_Data.Rz(detector_Euler(3))*Input_Data.Rx(detector_Euler(2))*Input_Data.Rz(detector_Euler(1));
g_tilt=Input_Data.Rx(double(dataset_header.Tilt_Angle));

%extract the orientation data
gmatrix_mtex=ebsd_original(pattern_number).orientations.matrix; %extract this from the ebsd data
gmatrix_detector=gmatrix_mtex'*g_tilt*g_camera'; %convert this to the detector frame

[Pattern_Sim]=EBSP_gen( EBSD_Geometry,gmatrix_detector,screen_int); %simulate the pattern
