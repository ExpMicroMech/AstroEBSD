% SINGLE PATTERN MATCH FOR TFS DATA WITH PHASE GROUP APPROACH
%
% Single pattern matching using refined template matching
% Runs natively on Thermo-Fisher Scientific EBSD data
% Includes dynamic plotting - automatically adjust final pattern plots
% according to the number of phase entries
%
% PHASE GROUP APPROACH
% First runs the SO(3) match on the first phase (recommended to be primary
% and fastest phase) - then splits points into 3 groups according to
% refined XCF.
% - group 1 = XCF>threshold, runs on second (easily confused) only
% - group 2 = minimum < XCF < threshold, runs on all phases
% - group 3 = not even a close match, skipping second phase, running on
% remaining phases
% Example case: Ni matrix, MC-carbide, M7C3-carbide
%
% KEY THINGS TO BE AWARE OF WHEN USING THIS CODE
% setting Threshold.MainPhase = will be dependent on your phases, test on
% more than one point to confirm what number to use!
%
% Updated 28/09/2025 by MTse to include in AstroEBSD v2

Begin Matching and Refinement

usePar=0; % use the parallel loop for the library testing (1 = use, 0 = disable)

% Initialize arrays to store results
G_SO3_n=zeros(3,3,n_phase);
G_Refined_SO3_n=G_SO3_n;

Library_PH_n=zeros(n_phase,1);
SO3_PH_n=Library_PH_n;

%Flag initiation - used to decide if phase2 should be skipped (don't change)
skip_phase_2 = 0;

% Logics begin here - careful when editing...
% Loop through each phase as usual

for n = 1: n_phase
    % Logic for bypassing phase 2
    % If n=2 and the skip flag is on, bypass ALL processing for n=2
    if n == 2 && skip_phase_2 == 1
        disp('Phase 2 skipped due to weak Phase 1 match. Proceeding to Phase 3...');
        skip_phase_2 = 0; % Reset the flag
        continue;         % Skips to n=3, bypassing the processing below
    end

    % PHASE PROCESSING (NOW CONDITIONAL, but only bypassed above)
    [G_SO3, Library_PH, ~, ~] = fLibraryTest(template_library_n{n}, library_G_n{n}, Pat_Ref_r.FFT, SettingsXCF, XCF_data_fill, usePar);
    [G_Refined_SO3, regout_RSO3] = refine5(Pat_Ref_r, EBSD_geom, EBSD_geom.PC, G_SO3, SettingsXCF, screen_int{n}, RTM_info(n).isHex, RTM);

    % Store the results
    G_SO3_n(:,:,n) = G_SO3;
    G_Refined_SO3_n(:,:,n) = G_Refined_SO3;
    Library_PH_n(n) = Library_PH;
    SO3_PH_n(n) = regout_RSO3(4); % 4th value is the normalized PH

    % Logic for the refinements
    if n == 1
        % --- LOGIC FOR THE FIRST PHASE (n=1) ---
        if SO3_PH_n(n) > Threshold.MainPhase
            disp(['Strong match for Phase 1 (Refined XCF = ' num2str(SO3_PH_n(n)) '). Running Phase 2 and then exiting.']);

        elseif SO3_PH_n(n) < Threshold.WrongPhase
            disp(['Weak match for Phase 1 (Refined XCF = ' num2str(SO3_PH_n(n)) '). Phase 2 will be skipped.']);
            skip_phase_2 = 1;

        else
            disp(['Moderate match for Phase 1 (Refined XCF = ' num2str(SO3_PH_n(n)) '). Continuing normally for all phases.']);
        end

        continue;

    elseif n == 2
        % LOGIC FOR THE SECOND PHASE (n=2)
        % The skip check is already done at the top of the loop.

        % 1. Check if the exit condition was set by Phase 1 (XCF > Threshold.MainPhase)
        if exist('SO3_PH_n', 'var') && numel(SO3_PH_n) >= 1 && SO3_PH_n(1) > Threshold.MainPhase
            disp('Phase 2 completed as mandated by Phase 1 condition (XCF > Threshold.MainPhase). Exiting loop now.');
            break; % Exit the loop after running Phase 2
        end

    elseif n >= 3
        % LOGIC FOR PHASES 3 and beyond - continue as normal if loop has
        % not been exited already
        continue

    end

end

disp('Pattern matching completed.')

Dynamic plotting

Plot will automatically adjust to the number of phase simulated

% Pre-allocate cell arrays for pattern simulations
PS_L = cell(1, n_phase);
PS_R = cell(1, n_phase);

% Generate all pattern simulations in a loop
for n = 1:n_phase
    PS_L{n} = EBSP_gen(EBSD_geom, G_SO3_n(:,:,n), screen_int{n});
    PS_R{n} = EBSP_gen(EBSD_geom, G_Refined_SO3_n(:,:,n), screen_int{n});
end

% Create the plot
figure;
fs = 8;

% Determine the number of rows and columns for the subplot grid
num_rows = 3; % Calculate rows needed for a 3-column layout
num_cols = n_phase;

% Plot the experimental pattern in the top-center subplot
subplot(num_rows, num_cols, 1);
pPattern(pattern1, EBSD_geom);
title('Experimental');

% Plot all the simulated patterns
% Loop to plot the Library and Refined patterns
for n = 1:n_phase
    % Calculate subplot indices based on a flattened grid
    lib_index = n_phase + n;
    ref_index = n_phase*2 + n;

    % Plot Library Patterns for the current phase (second row)
    subplot(num_rows, num_cols, lib_index);
    pPattern(PS_L{n}, EBSD_geom);
    title(['L-' InputUser.Phase_Input{n} ' 0kV PH= ' sprintf('%0.2g', Library_PH_n(n))],'FontSize',fs);

    % Plot Refined Patterns for the current phase (third row)
    subplot(num_rows, num_cols, ref_index);
    pPattern(PS_R{n}, EBSD_geom);
    title(['R-' InputUser.Phase_Input{n} ' 0kV PH= ' sprintf('%0.2g', SO3_PH_n(n))],'FontSize',fs);
end
