Now run a pattern against each phase

fprintf('%s: Map matching started.\n',text_out);

pattot = ebsd_header.TotalNumberOfPoints;
usePar=1; %use the parallel loop for the library testing


% Initialise data storage arrays
G_SO3_map=zeros(3,3,n_phase,ebsd_header.TotalNumberOfPoints);
G_Refined_SO3_map=G_SO3_map;
Library_PH_map=zeros(n_phase,ebsd_header.TotalNumberOfPoints);
SO3_PH_map=Library_PH_map;

tic

% MAIN LOOP FOR PATTERN MATCHING BEGINS HERE

% Pattern Loading in parallel or all at once
parfor patnum = 1:pattot
    % Pattern loading and preparation
    if PatLoadSingle == 1
        pattern_loc=[ebsd(patnum).x_in,ebsd(patnum).y_in];
        [loaded_TFS_patterns] = loadPat_TFS(TFSpath,pattern_loc,1);
        exppattern=loaded_TFS_patterns.pat_proc;
    else
        exppattern=loaded_TFS_patterns.pat_proc;
    end

    [ pattern1 ] = EBSP_BGCor( exppattern, Settings_CorX);
    [Pat_Ref_r,XCF_data_fill] = refine_prep(pattern1,SettingsXCF,RTM);

    % Temporary storage for current pattern
    G_SO3_n=zeros(3,3,n_phase);
    G_Refined_SO3_n=G_SO3_n;
    Library_PH_n=zeros(n_phase,1);
    SO3_PH_n=Library_PH_n;
    skip_phase_2 = 0; % Reset skip flag for each new pattern
    local_usePar = 0; % Set fLibraryTest to run serially within the main parfor loop (usually 0 is ok)




    % NEW LOGIC
    for n = 1:n_phase

        % Logic for bypassing phase 2
        if n == 2 && skip_phase_2 == 1
            skip_phase_2 = 0; % Reset the flag
            continue;         % Skips to n=3
        end

        % PHASE PROCESSING (fLibraryTest and refine5)
        [G_SO3, Library_PH, ~, ~] = fLibraryTest(template_library_n{n}, library_G_n{n}, Pat_Ref_r.FFT, SettingsXCF, XCF_data_fill, local_usePar);
        [G_Refined_SO3, regout_RSO3] = refine5(Pat_Ref_r, EBSD_geom, EBSD_geom.PC, G_SO3, SettingsXCF, screen_int{n}, RTM_info(n).isHex, RTM);

        % Store the results in the temporary 'n' arrays
        G_SO3_n(:,:,n) = G_SO3;
        G_Refined_SO3_n(:,:,n) = G_Refined_SO3;
        Library_PH_n(n) = Library_PH;
        SO3_PH_n(n) = regout_RSO3(4); % 4th value is the normalized PH

        % Logic for the refinements
        if n == 1 %Logic for phase 1
            if SO3_PH_n(n) > Threshold.MainPhase

            elseif SO3_PH_n(n) < Threshold.WrongPhase
                skip_phase_2 = 1;
            else

            end


        elseif n == 2 % Logic for phase 2
            % Check if the exit condition was set by Phase 1
            if SO3_PH_n(1) > Threshold.MainPhase
                break; % Exit the internal phase loop (n=1 to n_phase)
            end

        end
    end % end of phase refinement loop



    % Sorting the data into storage
    G_SO3_map(:,:,:,patnum) = G_SO3_n;
    G_Refined_SO3_map(:,:,:,patnum) = G_Refined_SO3_n;
    Library_PH_map(:,patnum) = Library_PH_n;
    SO3_PH_map(:,patnum) = SO3_PH_n;

    % Display progress - broken placement
    % if feedback_number*round(n/feedback_number) == n
    %         disp(['Pattern ' int2str(n) ' of ' int2str(pattot) ' patterns matched']);
    % end

end % end of patnum = 1:pattot

toc

% Sort the data by the highest PH for each phase
[maxPH, maxPhaseNum] = max(test_ph1, [], 1);

for patnum=1:pattot
    maxG(:,:,patnum) = test_G1(:,:,maxPhaseNum(patnum),patnum); %extract the best Gmatrix
    magG_Sample(:,:,patnum)=maxG(:,:,patnum)*Detector_tilt.'; %calculate this in the sample
    rot_new(patnum)=rotation.byMatrix(magG_Sample(:,:,patnum)'); %replace the rotation with the updated rotation, and use this rotation as an active/passive is important

Set IPFs (dynamically)

% Generate IPF Keys
for x = 1:n_phase
    % Get the current phase from the cell array
    current_phase = all_phase_names{x};

    % Access the ebsd data for the current phase
    current_ebsd = ebsd_plot(current_phase);

    key_name = ['ipfKey', num2str(x)];

    % The TSL key creation can be slightly different for the first phase
    % and the rest, so we handle that with an if/else block.
    if x == 1
       % The first ipfKey is created differently
        eval([key_name, ' = ipfTSLKey(current_ebsd);']);
    else
        % The subsequent ipfKeys use the .CS property
        eval([key_name, ' = ipfTSLKey(current_ebsd.CS);']);
    end
end

%% now generate the matched and refined IPF colour maps of the individual phases

directions = {vector3d.X, vector3d.Y, vector3d.Z};
direction_names = {'X', 'Y', 'Z'};

% Loop through each phase
for i = 1:length(all_phase_names)
    current_phase_name = all_phase_names{i};
    current_ebsd = ebsd(current_phase_name);
    current_ipfKey = eval(strcat('ipfKey',num2str(i)));

    % Loop through each direction (X, Y, Z) for the current phase
    for j = 1:length(directions)
        current_direction = directions{j};
        current_direction_name = direction_names{j};

        % Set the direction for the IPF key
        current_ipfKey.inversePoleFigureDirection = current_direction;

        % Generate the colors
        colors = current_ipfKey.orientation2color(current_ebsd.orientations);

        % Create the dynamic variable name (e.g., colors_X1i, colors_Y2i)
        variable_name = ['colors_', current_direction_name, num2str(i), 'i'];

        % Assign the colors to the dynamically named variable
        eval([variable_name, ' = colors;']);
    end
end
%}
