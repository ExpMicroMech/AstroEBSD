Now read a H5 file

InputUser.HDF5_folder='G:\Chris_Bilsland\Year_2'; %Change this to the file location in whch you have saved the example data

InputUser.HDF5_folder='E:\ChrisB';
InputUser.HDF5_file='ebsd_inconel_625.h5';
InputUser.Phase_Input  = {'Ni-Nickel-cubic.cif'};

OutputUser_HDF5_file=[InputUser.HDF5_file(1:end-3) '_Astro.h5'];

MTEX_settings.grainangle=5; %grain boundary in degrees
MTEX_settings.xpref='west';
MTEX_settings.ypref='outOfPlane';

read the CIF

cs=loadCIF(InputUser.Phase_Input{1});
cs_list={'notIndexed',cs};
cs_view=2; %phase for visualisation

Load the h5 to MTEX

disp('Loading h5 to MTEX');

%read the h5
[ MapData,MicroscopeData,PhaseData,EBSPData ] = bReadHDF5( InputUser );
[Data_InputMap] = EBSD_Map(MapData,MicroscopeData);

%convert to EBSD container
[ebsd] = h5_to_MTEX(Data_InputMap,MTEX_settings,cs_list);

%select a phase for visualisation
phase1=ebsd.CSList{cs_view}.mineral;

Select the region of interest

%Plot the IPF map
%Select the outline of the rectangle (must select the top left point first)
figure;
oM1=ipfHSVKey(ebsd(phase1));
hold on;
plot(ebsd(phase1), oM1.orientation2color(ebsd(phase1).orientations));

[x,y]=ginput(2);
%Create a region of interest with aspect ratio 1.5 from the selected points
%find the diff in x and y
xDiff = abs(x(1) - x(2));
yDiff = abs(y(1) - y(2));
aspectR = xDiff/yDiff;  %get the aspect ratio
if aspectR ~= 1.5 % if the aspect ratio is not correct the find the new difff adjusting by x and y step
    while xDiff/yDiff>1.5
    xDiff = xDiff - 0.1397;
    yDiff = yDiff + 0.1400;
    end
    while xDiff/yDiff <1.5
    xDiff = xDiff + 0.1397;
    yDiff = yDiff - 0.1400;
    end
    aspectR = xDiff/yDiff;
end
%create the correct bottom right point with the right aspect ratio
x(2) = x(1) - xDiff;
y(2) = y(1) + yDiff;
%define the new region in um
region = [x(2) y(1) (x(1)-x(2)) (y(2)-y(1))];

%Plot the overlaying image
close all
figure;
plot(ebsd(phase1), oM1.orientation2color(ebsd(phase1).orientations));
hold on
rectangle('position', region, 'edgecolor', 'r', 'linewidth', 2);
hold off

%extract the ebsd data from within the map
condition = inpolygon(ebsd, region);
%create the new data container
ebsd_new = ebsd(condition);

%plot only the new area
%calculate the grains from the EBSD data
[grains_new,ebsd_new.grainId,ebsd_new.mis2mean] = calcGrains(ebsd_new,'angle',MTEX_settings.grainangle*degree);
figure;
oM1=ipfHSVKey(ebsd_new(phase1));
hold on;
plot(ebsd_new(phase1), oM1.orientation2color(ebsd_new(phase1).orientations));

now extract the data we want for saving as a h5

%create the file container
OutputUser=InputUser;

%we assume that this is an h5 file, hdf5 will cause issues with the (end-3)
OutputUser.DataName=InputUser.HDF5_file(1:end-3);
OutputUser.HDF5_file=OutputUser_HDF5_file;
OutputUser.HDF5FullFile=fullfile(OutputUser.HDF5_folder,OutputUser.HDF5_file);

dtype='/EBSD/Data/'; %EBSD data location
htype='/EBSD/Header/'; %Header data location (e.g. microscope settings)

%read the properties
StructNames=fieldnames(ebsd_new.prop);
num_structures=numel(StructNames);

x_beam_unique=unique(ebsd_new.prop.x);
y_beam_unique=unique(ebsd_new.prop.y);
NCOLS=numel(x_beam_unique);
NROWS=numel(y_beam_unique);
NPoints=numel(ebsd_new.prop.x);

%write the structures
for n=1:num_structures
    if isnumeric(ebsd_new.prop.(StructNames{n})) %check this is an array that could be useful
        h5_WritePair(OutputUser.HDF5FullFile,OutputUser.DataName,dtype,StructNames{n},ebsd_new.prop.(StructNames{n})); %write the data
    end
end



%orientation data
Astro_Phi1=zeros(size(ebsd_new.prop.x));
Astro_Phi2=zeros(size(ebsd_new.prop.x));
Astro_PHI=zeros(size(ebsd_new.prop.x));

for p=1:NPoints
    try
        Astro_Phi1(p)=ebsd_new(p).orientations.phi1*180/pi;
        Astro_Phi2(p)=ebsd_new(p).orientations.phi2*180/pi;
        Astro_PHI(p)=ebsd_new(p).orientations.Phi*180/pi;
    catch
        Astro_Phi1(p)=-1;
        Astro_Phi2(p)=-1;
        Astro_PHI(p)=-1;
    end

end
[x_beam_new,y_beam_new]=meshgrid(1:NCOLS,1:NROWS);
x_beam_new_m=x_beam_new(:)-1; %change to 0 based index
y_beam_new_m=y_beam_new(:)-1; %change to 0 based index

%force some variable naming
h5_WritePair(OutputUser.HDF5FullFile,OutputUser.DataName,dtype,'X BEAM',x_beam_new_m); %write the data
h5_WritePair(OutputUser.HDF5FullFile,OutputUser.DataName,dtype,'Y BEAM',y_beam_new_m); %write the data
h5_WritePair(OutputUser.HDF5FullFile,OutputUser.DataName,dtype,'X SAMPLE',ebsd_new.prop.x); %write the data
h5_WritePair(OutputUser.HDF5FullFile,OutputUser.DataName,dtype,'Y SAMPLE',ebsd_new.prop.y); %write the data




h5_WritePair(OutputUser.HDF5FullFile,OutputUser.DataName,dtype,'PHI',Astro_PHI); %write the data
h5_WritePair(OutputUser.HDF5FullFile,OutputUser.DataName,dtype,'phi2',Astro_Phi2); %write the data
h5_WritePair(OutputUser.HDF5FullFile,OutputUser.DataName,dtype,'phi1',Astro_Phi1); %write the data

%write the header
%header
h5_WritePair(OutputUser.HDF5FullFile,OutputUser.DataName,htype,'CameraTilt',MicroscopeData.CameraTilt); %write the data
h5_WritePair(OutputUser.HDF5FullFile,OutputUser.DataName,htype,'SampleTilt',MicroscopeData.SampleTilt); %write the data
h5_WritePair(OutputUser.HDF5FullFile,OutputUser.DataName,htype,'TotalTilt',MicroscopeData.TotalTilt); %write the data
h5_WritePair(OutputUser.HDF5FullFile,OutputUser.DataName,htype,'KV',MicroscopeData.KV); %write the data

h5_WritePair(OutputUser.HDF5FullFile,OutputUser.DataName,htype,'XSTEP',MicroscopeData.XSTEP); %write the data
h5_WritePair(OutputUser.HDF5FullFile,OutputUser.DataName,htype,'YSTEP',MicroscopeData.YSTEP); %write the data

h5_WritePair(OutputUser.HDF5FullFile,OutputUser.DataName,htype,'Magnification',MicroscopeData.Magnification); %write the data
h5_WritePair(OutputUser.HDF5FullFile,OutputUser.DataName,htype,'WD',MicroscopeData.WD); %write the data

%update array sizes
h5_WritePair(OutputUser.HDF5FullFile,OutputUser.DataName,htype,'NCOLS',NCOLS); %write the data
h5_WritePair(OutputUser.HDF5FullFile,OutputUser.DataName,htype,'NROWS',NROWS); %write the data
h5_WritePair(OutputUser.HDF5FullFile,OutputUser.DataName,htype,'NPoints',NPoints); %write the data
