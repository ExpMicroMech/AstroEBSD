Calculate Z-scores (population normalised peak height)

%Nb this doesn't need RTM.KeepLib=1, the parameters are inferred during the RTM procedure anyway.
Z=NormaliseScores_lognorm(RTM.Output.PeakHeight,RTM.Output.pds);

%And can infer the inverse
PH=unNormaliseScores_lognorm(Z, RTM.Output.pds);

%So can get a cutoff score corresponding to Z=3,
% equivalent to a limit that Z-scores are > 3std away from mean;
PH_lim=unNormaliseScores_lognorm(3.*ones(size(Z)),RTM.Output.pds);

%Or more generically
%for a given probability of non-randomness, p:
%this computes the inverse normal cumulative distribution function
%to get the Z-score for a specified p-value
p=0.99;
Z_lim=norminv(p);

%then using the above:
PH_lim=unNormaliseScores_lognorm(Z_lim.*ones(size(Z)),RTM.Output.pds);

Some visualisations below:

plot histograms of true peak heights, and fitted lognormal dists

figure

for n=1:4

    subplot(1,4,n)

    histogram(log(RTM.Output.Lib{n}),30,'Normalization','pdf')
    hold on
    pd=makedist('normal',RTM.Output.pds{n}(1),RTM.Output.pds{n}(2));
    x=[-5:0.01:5];
    plot((x),pd.pdf(x),'LineWidth',2);

    max_ph=log(max(RTM.Output.Lib{n}));
    plot(max_ph,pd.pdf(max_ph),'x','MarkerSize',15)

    title(InputUser.Phases{n})
    xlim([-5,0])
    xlabel('Log XCF peak height')
    ylabel('Probability density')

    pbaspect([1,1,1])

    [h(n),p(n)]=chi2gof(log(RTM.Output.Lib{n}));

end
