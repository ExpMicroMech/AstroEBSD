Generate the SO(3) Library

[EBSD_geom ] = EBSP_Gnom( RTM, Settings_PCin.start);

library_G_n=cell(n_phase,1);
template_library_n=cell(n_phase,1);
cs_list_n=cell(n_phase+1,1);
cs_list_n{1}='notIndexed';
for n=1:n_phase
    %populate fundamental zone of SO(3)
    cs_phase=loadCIF(RTM_info(n).cif_file);
    [ library_G ] = SO3_rotmat_gen( cs_phase,RTM.Sampling_Freq);
    % generate the library
    [ template_library ] = Library_Gen(EBSD_geom,screen_int{n},RTM_info(n).isHex,library_G,2,SettingsXCF);

    %put these in the bigger cell
    library_G_n{n}=library_G;
    template_library_n{n}=template_library;
    cs_list_n{n+1}=cs_phase;
end

clear library_G template_library

[ text_out ] = pNameClock;

exppattern=allProcessedPatterns(:,:,patnum);

%prepare the pattern
[ pattern1 ] = EBSP_BGCor( exppattern, Settings_CorX);
[Pat_Ref_r,XCF_data_fill] = refine_prep(pattern1,SettingsXCF,RTM);

%now try all thress phases

usePar=1; %use the parallel loop for the library testing

G_SO3_n=zeros(3,3,n_phase);

Library_PH_n=zeros(n_phase,1);
SO3_PH_n=Library_PH_n;
tic
for n=1:n_phase
    [G_SO3,Library_PH,~,~]=fLibraryTest(template_library_n{n},library_G_n{n},Pat_Ref_r.FFT,SettingsXCF,XCF_data_fill,usePar);
    [G_Refined_SO3,regout_RSO3] = refine5(Pat_Ref_r,EBSD_geom,EBSD_geom.PC,G_SO3,SettingsXCF,screen_int{n},RTM_info(n).isHex,RTM);

    G_SO3_n(:,:,n)=G_SO3;
    G_Refined_SO3_n(:,:,n)=G_Refined_SO3;
    Library_PH_n(n)=Library_PH;
    SO3_PH_n(n)=regout_RSO3(4); %the 4th value of regout is the normalized PH
    toc
end

Pattern centre update

%move the pattern centre around on the map based upon the known geometry of
%the sample and the detector.
%
% the detector is 256 pixels in size and the pixel is 55 um in width
% the sample is at 70 degrees in tilt
PCx_update = @(x)Settings_PCin.start(1) - x * scan_step / (256 * 55);
PCy_update = @(y)Settings_PCin.start(2) - y * scan_step / (256 * 55) * cos(70*pi/180);
PCz_update = @(y)Settings_PCin.start(3) + y * scan_step / (256 * 55) * sin(70*pi/180);
