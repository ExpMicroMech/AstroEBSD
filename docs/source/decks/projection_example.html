Simulate a single pattern

PatternInfo.ScreenHeight=250;
PatternInfo.ScreenWidth=250;
PC=[0.5 0.3 0.6]; %PCx, PCy, DD %which way is up/Bruker convention

% PC=[0.7 0.39 0.19]; %PCx, PCy, DD
% eangs=[15,37,45]; %in degrees
eangs=[-106,177,180+12]; %in degrees
% gmatrix_mtex=eye(3);
gmatrix=Input_Data.Rz(eangs(3)*degree)*Input_Data.Rx(eangs(2)*degree)*Input_Data.Rz(eangs(1)*degree);


[EBSD_geom ] = EBSP_Gnom( PatternInfo,PC); %you can change PC_in if you want
[Pattern_Sim]=EBSP_gen( EBSD_geom,gmatrix*g_tilt*g_camera',screen_int);

figure;
pPattern(Pattern_Sim,EBSD_geom);

Simulate a sphere

figure;
[x,y,z]=sphere(2000);
[ebsp_i2] = Cube_Sample(x,y,z,screen_int,screen_int.isHex);
ebsp_i2=reshape(ebsp_i2,size(x,1),size(x,2));
surf(x,y,z,ebsp_i2,'EdgeColor','none');
axis equal; colormap('gray');
xlabel('X'); ylabel('Y'); zlabel('Z');
view(0,0)
