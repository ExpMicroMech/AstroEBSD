Now compare a pattern with this to index and refine

tic

patnum=23021;
exppattern=allProcessedPatterns(:,:,patnum);

PCin = Settings_PCin.start;
[EBSD_geom ] = EBSP_Gnom( RTM, PCin);

%prepare the pattern
[ pattern1 ] = EBSP_BGCor( exppattern, Settings_CorX);
% [ pattern1 ] = EBSP_BGCor( thisProcdPattern, []);
[Pat_Ref_r,XCF_data_fill] = refine_prep(pattern1,SettingsXCF,RTM);

%check one pattern by plotting it
% figure; pPattern(pattern1,EBSD_geom)

%find the nearest SO(3)

usePar=1; %use the parallel loop for the library testing
[G_SO3,Library_PH,~,~]=fLibraryTest(template_library,library_G,Pat_Ref_r.FFT,SettingsXCF,XCF_data_fill,usePar);
toc
%refine based upon the SO3 found solution
[G_Refined_SO3,regout_RSO3] = refine5(Pat_Ref_r,EBSD_geom,EBSD_geom.PC,G_SO3,SettingsXCF,screen_int,RTM_info.isHex,RTM);

%refine this pattern
Detector_tilt = RTM.Rx(ebsd_header.SpecimenTilt-pi/2);
eangs=[ebsd(patnum).prop.Phi1,ebsd(patnum).prop.PHI,ebsd(patnum).prop.Phi2]*pi/180;

%TFS refine
gmatrix=RTM.Rz(eangs(3))*RTM.Rx(eangs(2))*RTM.Rz(eangs(1));
[G_Refined_TFS,regout_RTFS] = refine5(Pat_Ref_r,EBSD_geom,EBSD_geom.PC,gmatrix*Detector_tilt,SettingsXCF,screen_int,RTM_info.isHex,RTM);


%generate the simulated patterns to check solutions
[ Pat_sim_SO3 ] = EBSP_gen( EBSD_geom,G_SO3,screen_int); %generate the EBSP for this iteration
[ Pat_sim_SO3_refined ] = EBSP_gen( EBSD_geom,G_Refined_SO3,screen_int); %generate the EBSP for this iteration
[ Pat_sim_TFS_refined ] = EBSP_gen( EBSD_geom,G_Refined_TFS,screen_int); %generate the EBSP for this iteration


[ Pat_sim_TFS ] = EBSP_gen( EBSD_geom, gmatrix*Detector_tilt,screen_int); %generate the EBSP for this iteration

toc

% Plot the results
figure;
nexttile;
pPattern(pattern1,EBSD_geom); title('Experimental')
nexttile;
pPattern(Pat_sim_SO3,EBSD_geom); title('Sim - SO(3)')
nexttile;
pPattern(Pat_sim_SO3_refined,EBSD_geom); title('Sim - SO(3) - refined')
nexttile;
pPattern(Pat_sim_TFS,EBSD_geom); title('TFS Hough')
nexttile;
pPattern(Pat_sim_TFS_refined,EBSD_geom); title('TFS Refined')

quality_values=[Library_PH regout_RTFS(4) regout_RSO3(4)]; %extract the quality metrics and have a look if you want

% Take these EBSD values and put back into an EBSD container

Gs_Refined_SO3=G_Refined_SO3*Detector_tilt.'; %use .' as the real transpose, which is the inverse as this is a rotation matrix
rot=rotation.byMatrix(Gs_Refined_SO3'); %use this rotation as an active/passive is important

prop_new=ebsd(patnum).prop;
prop_new.XCF=regout_RSO3(4);
prop_new.Phi1_old=prop_new.Phi1;
prop_new.Phi2_old=prop_new.Phi2;
prop_new.PHI_old=prop_new.PHI;
prop_new.Phi1=rot.phi1./degree;
prop_new.Phi2=rot.phi2./degree;
prop_new.PHI=rot.Phi./degree;

ebsdtemp = EBSD(rot,1,cs_phase,prop_new,'unitCell',calcUnitCell([prop_new.y,prop_new.x]));

angle(ebsdtemp.orientations,ebsd(patnum).orientations)*180/pi;

Pattern centre update

%move the pattern centre around on the map based upon the known geometry of
%the sample and the detector.
%
% the detector is 256 pixels in size and the pixel is 55 um in width
% the sample is at 70 degrees in tilt
PCx_update = @(x)Settings_PCin.start(1) - x * scan_step / (256 * 55);
PCy_update = @(y)Settings_PCin.start(2) - y * scan_step / (256 * 55) * cos(70*pi/180);
PCz_update = @(y)Settings_PCin.start(3) + y * scan_step / (256 * 55) * sin(70*pi/180);
