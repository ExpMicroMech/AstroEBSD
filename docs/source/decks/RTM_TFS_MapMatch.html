[ text_out ] = pNameClock;
fprintf('%s: Pattern library generating.\n',text_out);

[EBSD_geom ] = EBSP_Gnom( RTM, Settings_PCin.start);

library_G_n=cell(n_phase,1);
template_library_n=cell(n_phase,1);
cs_list_n=cell(n_phase+1,1);
cs_list_n{1}='notIndexed';
for n=1:n_phase
    %populate fundamental zone of SO(3)
    cs_phase=loadCIF(RTM_info(n).cif_file);
    [ library_G ] = SO3_rotmat_gen( cs_phase,RTM.Sampling_Freq);
    % generate the library
    [ template_library ] = Library_Gen(EBSD_geom,screen_int{n},RTM_info(n).isHex,library_G,2,SettingsXCF);

    %put these in the bigger cell
    library_G_n{n}=library_G;
    template_library_n{n}=template_library;
    cs_list_n{n+1}=cs_phase;
end

Detector_tilt = RTM.Rx(ebsd_header.SpecimenTilt-pi/2); %use this to convert between the EBSD detector and sample frames

maxG=zeros(3,3,pattot); %highest g matrix, SO(3) refined, from the different phase correlations
magG_Sample=maxG; %convert this into the sample frame

%subsetting this for our demo
test_ph1=SO3_PH_map(:,1:pattot);
test_G1=G_Refined_SO3_map(:,:,:,1:pattot);

rot_new=ebsd(1:pattot).rotations; %take the ebsd container, and use this to populate the rotations
prop_new=ebsd(1:pattot).prop; %same with the properties

%preallocation arrays
propt_XCF=zeros(size(prop_new.x));
propt_Phi1_old=propt_XCF;
propt_Phi2_old=propt_XCF;
propt_PHI_old=propt_XCF;

propt_Phi1=propt_XCF;
propt_Phi2=propt_XCF;
propt_Phi=propt_XCF;

% Sort the data by the highest PH for each phase
[maxPH, maxPhaseNum] = max(test_ph1, [], 1);

for patnum=1:pattot
    maxG(:,:,patnum) = test_G1(:,:,maxPhaseNum(patnum),patnum); %extract the best Gmatrix
    magG_Sample(:,:,patnum)=maxG(:,:,patnum)*Detector_tilt.'; %calculate this in the sample
    rot_new(patnum)=rotation.byMatrix(magG_Sample(:,:,patnum)'); %replace the rotation with the updated rotation, and use this rotation as an active/passive is important

    propt_XCF(patnum)=maxPH(patnum);

    %there is something odd about this carrying forward - we could not fix
    %at the time - multiple euler angles were returned

    % propt_Phi1_old(patnum)=prop_new(patnum).Phi1(1);
    % propt_Phi2_old(patnum)=prop_new(patnum).Phi2(1);
    % propt_PHI_old(patnum)=prop_new(patnum).PHI(1);

    propt_Phi1(patnum)=rot_new(patnum).phi1./degree;
    propt_Phi2(patnum)=rot_new(patnum).phi2./degree;
    propt_Phi(patnum)=rot_new(patnum).Phi./degree;
end

%put these into the properties structure
