load all the patterns, and count the MEC

pattern_loc=[ebsd.x_in,ebsd.y_in]; %note this is in 0 based indexing, whole map
% [patterns_full] = loadPat_TFS(TFS_DataLoc,[1,1;1,2],2); %loadPat_TFS([file location],[list of patterns, 0 indexed, two colums x,y], [1 = procssed, 2 = raw, 3 = both])
tic
[patterns_full] = loadPat_TFS(TFS_DataLoc,pattern_loc,3); %loadPat_TFS([file location],[list of patterns, 0 indexed, two colums x,y], [1 = procssed, 2 = raw, 3 = both])
toc
npats=size(patterns_full.pat_raw,3);

%calculate the MEC from this data
ebsd_mec=zeros(npats,1);
ebsd_mec_proc=zeros(npats,1);

fsd_r=zeros(npats,1); fsd_g=fsd_r;fsd_b=fsd_r;
fsd_rgb=zeros(npats,3);
for n=1:npats
    pattern_n=patterns_full.pat_raw(:,:,n);
    ebsd_mec(n)=median(pattern_n(:));

    pattern_p=patterns_full.pat_proc(:,:,n);
    ebsd_mec_proc(n)=median(pattern_p(:));

    %FSD coloring - this probably needs some work
    fsd_r(n)=sum(sum(pattern_n(1:85,1:26)));
    fsd_g(n)=sum(sum(pattern_n(86:170,1:26)));
    fsd_b(n)=sum(sum(pattern_n(171:256,1:26)));
    fsd_rgb(n,:)=[fsd_r(n),fsd_r(n),fsd_r(n)];
end

%add the MEC to the EBSD container
ebsd.prop.MEC=double(ebsd_mec);
ebsd.prop.MEC2=double(ebsd_mec_proc);
ebsd.prop.fsd_rgb=double(fsd_rgb);


%normalize rgb - this is very quick and dirty
fsd_r=0.5+(fsd_r-nanmean(fsd_r(:)))./nanstd(fsd_r(:));
fsd_g=0.5+(fsd_g-nanmean(fsd_g(:)))./nanstd(fsd_g(:));
fsd_b=0.5+(fsd_b-nanmean(fsd_b(:)))./nanstd(fsd_b(:));

%put int the container
fsd=[fsd_r,fsd_g,fsd_b];

ebsd.prop.fsd_r=double(fsd_r);
ebsd.prop.fsd_g=double(fsd_g);
ebsd.prop.fsd_b=double(fsd_b);

ebsd.prop.fsd=fsd;

%plot the MEC data
figure;
plot(ebsd,ebsd.prop.MEC,'micronbar','on');
nextAxis;
plot(ebsd,ebsd.prop.MEC2,'micronbar','off');
nextAxis;
plot(ebsd,ebsd.prop.fsd,'micronbar','off');
