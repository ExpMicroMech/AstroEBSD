Hough indexing for TFS Data - multiphase

This code is for indexing TFS pattern data using Hough Indexing. Points with high error values can then be selected to run refined template matching.

Designed to read patterns squentially by row instead of all together to reduce RAM load.

Key things to note: - Remember to update starting euler and PC - Check the .pha file you use has the correct $PlaneReflector and $Atoms (These may not have been updated if the original phase file was only meant to run RTM)

Go through the patterns

eangs = eangs_start;
gmatrix=RTM.Rz(eangs(3))*RTM.Rx(eangs(2))*RTM.Rz(eangs(1));
PCin = Settings_PCin.start;
% Detector_tilt = RTM.Rx(-20*pi/180); % for EBSD
Detector_tilt = eye(3); % for TKD - may need to adjust if using detetcor ECP
n_phase = size(InputUser.Phase_Input,2);

% index = 1;
for i=0:scan_height-1 % Loop for rows begin here
    paddedRowNumber = sprintf( '%05d', (i) );
    binFileName = strcat("row_", paddedRowNumber, ".bin");
    fullBinFileName = fullfile(TFSpath, 'patterns/' , binFileName);

    fileID = fopen(fullBinFileName, 'r');
    data =  fread(fileID);
    fclose(fileID);

    % Sanity check that the size of things match
    if mod(size(data,1), EBSDPatArraySize) ~=0
    error("length of pattern file not dividable by pattern size");
    end

    % Read patterns of this row (code for reading the .bin file remains the same)
    allPatsThisRow = reshape(data, [EBSDPatSize EBSDPatSize size(data,1)/EBSDPatArraySize]);
    fprintf('%s: Starting %d of %d row\n',datestr(now,'HH:MM:SS'), i+1, scan_height);
    clear data;

    % NEW: Initialize temporary row-based storage for new results
    Res_Phase_ThisRow = zeros(1, scan_width);
    Res_AllPhaseErrors_ThisRow = zeros(scan_width, n_phase);

    for j=0:scan_width-1 %Loop through every pattern in this row

        index = i * scan_width + j;

        Res_X_ThisRow(j+1) = j;
        Res_Y_ThisRow(j+1) = i;
        Res_ID_ThisRow(j+1) = index + 1;

        pattern_read = flipud(double(allPatsThisRow(:,:,j+1)))';

        % Pattern Center operations (PC calculation, background correction, PC search)
        PCin = Settings_PCin;
        PCin.start = [ Settings_PCin.start(1) - j * scan_step / (256 * 55);
            Settings_PCin.start(2) - i * scan_step / (256 * 55) * cos(70*pi/180);
            Settings_PCin.start(3) + i * scan_step / (256 * 55) * sin(70*pi/180)];

        [EBSD_geom ] = EBSP_Gnom( RTM, PCin.start); % At this point the PC should be updated already


        % Build the geometry + prepare the pattern
        EBSP_One=struct;
        [EBSP_One.PatternIn,~ ] = EBSP_BGCor( pattern_read, Settings_CorX);

        %[EBSP_One_PC_S] = EBSP_PCSearch(EBSP_One,Settings_CorX,Settings_Rad, PCin,Phase_Num,Crystal_LUT,Crystal_UCell);
        % Pass the total number of phases (n_phase) instead of the list (Phase_Num)

        % Find the Patter Center
        [EBSP_One_PC_S] = EBSP_PCSearch(EBSP_One,Settings_CorX,Settings_Rad, PCin,n_phase,Crystal_LUT,Crystal_UCell);
        % Index Normally - transform EBSP and find peaks
        [ EBSP_One.Peak_Centre,EBSP_One.Single.Peak_Set_All,EBSP_One.Peak_Set_All, ...
            EBSP_One.R_EBSP,EBSP_One.R_Edge,EBSP_One.R_rho,EBSP_One.R_theta ] ...
            = EBSP_RadHunt( EBSP_One.PatternIn,Settings_Rad);
        % Convert the bands to normal space
        [ EBSP_One.nhat_gnom] = EBSP_NormConv( EBSP_One.Peak_Centre,size(EBSP_One.PatternIn), EBSP_One_PC_S.PC_out);

        % Now we have the transformed EBSP with peaks identified
        % NEW: Multi-phase indexing section

        best_error = inf; % Initialize with a very high error
        best_phase_idx = 0; % Index for the best phase
        best_rotdata = []; % Structure to hold the best solution
        all_errors = inf(1, n_phase); % Array to hold errors for all phases

        % Loop through each phase
        for p = 1:n_phase
            % Index using the data for phase 'p'
            [rotdata_p, ~] = EBSP_Index(EBSP_One.nhat_gnom, Crystal_LUT{p}, Settings_LUT{p}.thresh_trig, Crystal_UCell{p}, Detector_tilt);

            % If a solution was found, store its error
            if ~isempty(rotdata_p)
                all_errors(p) = rotdata_p.error;

                % Check if this solution is better than the previous best
                if rotdata_p.error < best_error
                    best_error = rotdata_p.error;
                    best_phase_idx = p;
                    best_rotdata = rotdata_p;
                end
            end
        end

        % Store the errors for all phases for this point
        Res_AllPhaseErrors_ThisRow(j+1, :) = all_errors;

        % After checking all phases, save the results from the BEST one
        if ~isempty(best_rotdata)
            Res_EulerZ1_ThisRow(j+1) = best_rotdata.eang(1);
            Res_EulerX_ThisRow(j+1)  = best_rotdata.eang(2);
            Res_EulerZ2_ThisRow(j+1) = best_rotdata.eang(3);
            Res_XCF_ThisRow(j+1)     = best_rotdata.error;
            Res_Phase_ThisRow(j+1)   = best_phase_idx;
        else
            % If no phase produced a solution, save as zero or NaN
            Res_EulerZ1_ThisRow(j+1) = 0;
            Res_EulerX_ThisRow(j+1)  = 0;
            Res_EulerZ2_ThisRow(j+1) = 0;
            Res_XCF_ThisRow(j+1)     = inf;
            Res_Phase_ThisRow(j+1)   = 0;
        end
    end

    % start from rownum * rowpix (i) * rowsize + 1
    % end at (i+1) * rowsize
    thisRowStart = (i) * scan_width + 1;
    thisRowEnd = (i+1) * scan_width;

    Results.ID(thisRowStart:thisRowEnd) = Res_ID_ThisRow;
    Results.X(thisRowStart:thisRowEnd) = Res_X_ThisRow;
    Results.Y(thisRowStart:thisRowEnd) = Res_Y_ThisRow;
    Results.EulerZ1(thisRowStart:thisRowEnd) = Res_EulerZ1_ThisRow;
    Results.EulerX(thisRowStart:thisRowEnd) = Res_EulerX_ThisRow;
    Results.EulerZ2(thisRowStart:thisRowEnd) = Res_EulerZ2_ThisRow;
    Results.XCF(thisRowStart:thisRowEnd) = Res_XCF_ThisRow; % Storing the best error

    % NEW: Store the new multi-phase results
    Results.Phase(thisRowStart:thisRowEnd) = Res_Phase_ThisRow;
    Results.AllPhaseErrors(thisRowStart:thisRowEnd, :) = Res_AllPhaseErrors_ThisRow;

    fprintf('%s: %d of %d rows done\n',datestr(now,'HH:MM:SS'),i+1, scan_height);
end
